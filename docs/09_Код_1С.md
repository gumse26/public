# Код 1С

Гум-сервис для работы с ККТ через шлюз `gate.atol.piot`, по соглашениям написания
кода: локальные переменные — `л`, неэкспортные функции — `м`, экспортные — без префикса,
действия в 1–3 строки не выносятся, к элементам соответствия обращаемся через `Получить()`.

Отправитель команд в шлюз — очередь GATE (см. `00_Как_пользоваться.md`). Шлюз слушает
её в режиме `2_gateway_queue.bat`, принимает один JSON на команду и возвращает JSON-ответ.

## Модуль гум-сервиса

```1c
// Гум-сервис ККТ: команды шлюзу gate.atol.piot через очередь GATE.

// Адрес и очередь — из настроек; здесь для наглядности.
Функция Вызвать(М, П, К) Экспорт

	лРезультат = Неопределено;

	Если      М = "Состояние"               Тогда лРезультат = мСостояние(П, К);
	ИначеЕсли М = "ОткрытьСмену"            Тогда лРезультат = мОткрытьСмену(П, К);
	ИначеЕсли М = "Продажа"                 Тогда лРезультат = мЧек(П, К, "Продажа");
	ИначеЕсли М = "Возврат"                 Тогда лРезультат = мЧек(П, К, "Возврат");
	ИначеЕсли М = "ПродажаМаркированного"   Тогда лРезультат = мПродажаМаркированного(П, К);
	ИначеЕсли М = "СнятьОтчетБезГашения"    Тогда лРезультат = мОтчетБезГашения(П, К);
	ИначеЕсли М = "СнятьОтчетСГашением"     Тогда лРезультат = мОтчетСГашением(П, К);
	ИначеЕсли М = "ОткрытьЯщик"             Тогда лРезультат = мОткрытьЯщик(П, К);
	ИначеЕсли М = "ОтменитьЧек"             Тогда лРезультат = мОтменитьЧек(П, К);
	ИначеЕсли М = "ПроверитьКодыМаркировки" Тогда лРезультат = мПроверитьКодыМаркировки(П, К);
	Иначе
		ВызватьИсключение("Метод " + М + " не реализован");
	КонецЕсли;

	Возврат лРезультат;

КонецФункции

//---------------------------------------------------------------------------------------------------
//
//---------------------------------------------------------------------------------------------------

// Отправка одной команды шлюзу. П — параметры операции, лМетод — имя операции в протоколе.
Функция мВыполнить(П, К, лМетод)

	лДанные = Новый Соответствие;
	лДанные.Вставить(лМетод, П);

	лТело = мВJSON(лДанные);

	лСоединение = Новый HTTPСоединение(К.Сервер, К.Порт, , , К.Таймаут);
	лЗапрос = Новый HTTPЗапрос("/qst/" + К.Очередь + "/request/" + К.Ттл);
	лЗапрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
	лЗапрос.УстановитьТелоИзСтроки(лТело, КодировкаТекста.UTF8);

	лОтвет = лСоединение.ВызватьHTTPМетод("POST", лЗапрос);
	Если лОтвет.КодСостояния <> 200 Тогда
		ВызватьИсключение("Шлюз ответил: " + лОтвет.КодСостояния);
	КонецЕсли;

	лТекст = лОтвет.ПолучитьТелоКакСтроку(КодировкаТекста.UTF8);
	Если ПустаяСтрока(лТекст) Тогда
		ВызватьИсключение("Шлюз не ответил за " + К.Ттл + " с");
	КонецЕсли;

	// Разбор ответа: строки — через Получить(), как принято для соответствий.
	лЧтение = Новый ЧтениеJSON;
	лЧтение.УстановитьСтроку(лТекст);
	лРезультат = ПрочитатьJSON(лЧтение);

	Если лРезультат.Получить("Ошибка") <> Неопределено Тогда
		ВызватьИсключение(лРезультат.Получить("Ошибка"));
	КонецЕсли;

	Возврат лРезультат;

КонецФункции

Функция мВJSON(лЗначение)

	лЗапись = Новый ЗаписьJSON;
	лЗапись.УстановитьСтроку();
	ЗаписатьJSON(лЗапись, лЗначение);

	Возврат лЗапись.Закрыть();

КонецФункции

Функция мСостояние(П, К)

	Возврат мВыполнить(П, К, "Состояние");

КонецФункции

Функция мОткрытьСмену(П, К)

	Возврат мВыполнить(П, К, "ОткрытьСмену");

КонецФункции

// Общий чек: продажа и возврат отличаются только полем ТипЧека.
Функция мЧек(П, К, лТип)

	лЧек = Новый Соответствие;
	лЧек.Вставить("ТипЧека", лТип);
	лЧек.Вставить("СНО", П.Получить("СНО"));
	лЧек.Вставить("Кассир", мКассир(П));
	лЧек.Вставить("Оплата", мОплата(П.Получить("ОплатаНаличными"), П.Получить("ОплатаБезналичными")));
	лЧек.Вставить("Товары", мТовары(П.Получить("Товары")));
	лЧек.Вставить("СуммаСкидкиНаЧек", 0);

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("НапечататьЧек", лЧек);

	Возврат мВыполнить(лПараметры, К, "НапечататьЧек");

КонецФункции

Функция мКассир(П)

	лКассир = Новый Соответствие;
	лКассир.Вставить("Имя", П.Получить("Кассир"));

	Возврат лКассир;

КонецФункции

// Оплата — массив из 9 чисел: [0] нал, [1] безнал, остальные нули.
Функция мОплата(лНаличными, лБезналичными)

	Возврат мНовыйМассив(9, лНаличными, лБезналичными);

КонецФункции

// Товары из таблицы значений: Наименование, Количество, Цена, Сумма, СтавкаНДС, КодМаркировки.
Функция мТовары(лТаблица)

	лТовары = Новый Массив;

	Для Каждого лСтрока Из лТаблица Цикл
		лПозиция = Новый Соответствие;
		лПозиция.Вставить("Наименование", лСтрока.Наименование);
		лПозиция.Вставить("Количество", лСтрока.Количество);
		лПозиция.Вставить("Цена", лСтрока.Цена);
		лПозиция.Вставить("Сумма", лСтрока.Сумма);
		лПозиция.Вставить("Скидка", 0);
		лПозиция.Вставить("Подразделение", 0);
		лПозиция.Вставить("СтавкаНДС", лСтрока.СтавкаНДС);
		лПозиция.Вставить("НомерНалога", лСтрока.СтавкаНДС);
		лПозиция.Вставить("ПризнакСпособаРасчета", 4);

		// Маркированный товар: признак 33, код в Base64 и отраслевой реквизит.
		Если ЗначениеЗаполнено(лСтрока.КодМаркировки) Тогда
			лПозиция.Вставить("ПризнакПредметаРасчета", 33);
			лПозиция.Вставить("КодМаркировки", мBase64(лСтрока.КодМаркировки));
			лПозиция.Вставить("ОтраслевойРеквизитПредметаРасчета", мОтраслевой(лСтрока));
		Иначе
			лПозиция.Вставить("ПризнакПредметаРасчета", 1);
		КонецЕсли;

		лТовары.Добавить(лПозиция);
	КонецЦикла;

	Возврат лТовары;

КонецФункции

// Код маркировки: разделители GS1 (0x1D) уже стоят в строке, кодируем всё в Base64.
Функция мBase64(лКод)

	Возврат Base64Строка(ПолучитьДвоичныеДанныеИзСтроки(лКод, КодировкаТекста.UTF8));

КонецФункции

// Отраслевой реквизит: UUID из ответа проверки КМ и время сценария в миллисекундах.
Функция мОтраслевой(лСтрока)

	лРеквизит = Новый Соответствие;
	лРеквизит.Вставить("ИдентификаторФОИВ", "030");
	лРеквизит.Вставить("ДатаДокументаОснования", Формат(ТекущаяДата(), "ДЛФ=DD"));
	лРеквизит.Вставить("НомерДокументаОснования", "1");
	лРеквизит.Вставить("ЗначениеОтраслевогоРеквизита",
		"UUID=" + лСтрока.GUIDПроверки + "&Time=" + мВремяВМиллисекундах());

	Возврат лРеквизит;

КонецФункции

Функция мВремяВМиллисекундах()

	Возврат Цел(ТекущаяУниверсальнаяДатаВМиллисекундах());

КонецФункции

Функция мПродажаМаркированного(П, К)

	// Сначала проверка КМ, затем чек — и обязательно на одном и том же хендле устройства.
	лПроверка = мПроверитьКодыМаркировки(П, К);
	П.Вставить("GUIDПроверки", лПроверка.Получить("GUID"));

	Возврат мЧек(П, К, "Продажа");

КонецФункции

Функция мПроверитьКодыМаркировки(П, К)

	лЗапрос = Новый Соответствие;
	лЗапрос.Вставить("ЭтоВозврат", П.Получить("ЭтоВозврат"));
	лЗапрос.Вставить("ТипПодтверждения", 0);
	лЗапрос.Вставить("КодыМаркировки", П.Получить("КодыМаркировки"));

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("ПроверитьКодыМаркировки", лЗапрос);

	Возврат мВыполнить(лПараметры, К, "ПроверитьКодыМаркировки");

КонецФункции

Функция мОтчетБезГашения(П, К)

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("Пароль", П.Получить("Пароль"));

	Возврат мВыполнить(лПараметры, К, "СнятьОтчетБезГашения");

КонецФункции

Функция мОтчетСГашением(П, К)

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("Пароль", П.Получить("Пароль"));

	Возврат мВыполнить(лПараметры, К, "СнятьОтчетСГашением");

КонецФункции

Функция мОткрытьЯщик(П, К)

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("Пароль", П.Получить("Пароль"));

	Возврат мВыполнить(лПараметры, К, "ОткрытьЯщик");

КонецФункции

Функция мОтменитьЧек(П, К)

	лПараметры = Новый Соответствие;
	лПараметры.Вставить("Пароль", П.Получить("Пароль"));

	Возврат мВыполнить(лПараметры, К, "ОтменитьЧек");

КонецФункции
```

## Вызов из 1С

```1c
лК = Новый Соответствие;
лК.Вставить("Сервер", "gate.gumse.ru");
лК.Вставить("Порт", 47005);
лК.Вставить("Очередь", "kkt.test");
лК.Вставить("Ттл", 60);
лК.Вставить("Таймаут", 60);

лТовары = Новый ТаблицаЗначений;
лТовары.Колонки.Добавить("Наименование");
лТовары.Колонки.Добавить("Количество");
лТовары.Колонки.Добавить("Цена");
лТовары.Колонки.Добавить("Сумма");
лТовары.Колонки.Добавить("СтавкаНДС");
лТовары.Колонки.Добавить("КодМаркировки");

лСтрока = лТовары.Добавить();
лСтрока.Наименование = "Тестовая позиция";
лСтрока.Количество   = 1;
лСтрока.Цена         = 1;
лСтрока.Сумма        = 1;
лСтрока.СтавкаНДС    = "5";

лП = Новый Соответствие;
лП.Вставить("СНО", 1);
лП.Вставить("Кассир", "Администратор");
лП.Вставить("ОплатаНаличными", 1);
лП.Вставить("ОплатаБезналичными", 0);
лП.Вставить("Товары", лТовары);

лЧек = Вызвать("Продажа", лП, лК);
Сообщить("Чек " + лЧек.Получить("НомерЧека") + ", ФД " + лЧек.Получить("НомерДокумента")
	+ ", ФП " + лЧек.Получить("ФискальныйПризнак"));
```

## Прямой вызов компоненты (для сравнения)

Шлюз делает ровно то же самое, но по HTTP. Если вызывать компоненту из 1С напрямую,
порядок такой: `ПараметрыОборудования` → `ПодключитьОборудование` → рабочая команда.
XML — тот же, что в дампах (`01_Продажа.md` и далее).

```1c
лВК = Новый("AddIn.ATOL_KKT_1C_V10_X_ESM");

лИД = "";
лВК.ПодключитьОборудование(лИД, "ККТ", лXmlНастроек);   // ИД устройства — выходной

лУспех = Ложь;
лXml = "";
лВК.СформироватьЧек(лИД, Ложь, лXmlЧека, лXml);         // 4-й параметр — размер буфера/строка
лУспех = лВК.ПолучитьТекущееСостояние(лИД, лXml);
```

**Важно:** хендл устройства, полученный в `ПодключитьОборудование`, надо использовать
для **всех** последующих команд. Проверка КМ и чек, выполненные на разных хендлах, дают
ошибку `420 «Контрольная марка не проверена»`.
